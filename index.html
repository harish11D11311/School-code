Program 2

#program to calculate amount to be payed(simpleinterest)
p=int(input("enter the principal amount"))
n=float(input("enter the no of years"))
r=float(input("enter the rate of interest"))
si=(p*n*r)/100
print("amount payed=",p+si)

Program 3

#program to calculate amount to be payed(compound interest)
p=int(input("enter the principal amount"))
n=float(input("enter the no of years"))
r=float(input("enter the rate of interest"))
ci=(p((1+r/100)**n))
print("amount payed=",p+ci)
